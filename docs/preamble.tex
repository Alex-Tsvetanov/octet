% ============================================================================
% preamble.tex: formatting by the rules of Prof. G. Iliev for this course project:
% Times New Roman 12 or 14 pt (14 chosen), line spacing 1.5, justified text,
% numbered pages, Bulgarian. Engine: pdfLaTeX with T2A and tempora (a Times clone
% with full Cyrillic).
% ============================================================================

\usepackage{cmap}
\usepackage[T2A]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[bulgarian]{babel}
\usepackage{tempora}
\usepackage[a4paper, top=2cm, bottom=2.2cm, left=2cm, right=2cm, footskip=1cm]{geometry}
\usepackage{setspace}
\setstretch{1.5}
\usepackage[expansion=false]{microtype}
\usepackage{indentfirst}
\setlength{\parindent}{1.25cm}
\setlength{\parskip}{0pt}

% Page numbers, bottom centre.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\thepage}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\thepage}}

% Headings: arabic numbering, bold.
\usepackage{titlesec}
\titleformat{\section}{\fontsize{16}{19}\selectfont\bfseries}{\thesection.}{0.5em}{}
\titleformat{\subsection}{\fontsize{14}{17}\selectfont\bfseries}{\thesubsection.}{0.5em}{}
\titleformat{\subsubsection}{\fontsize{14}{17}\selectfont\bfseries\itshape}{\thesubsubsection.}{0.5em}{}
\titlespacing*{\section}{0pt}{1.2em}{0.6em}
\titlespacing*{\subsection}{0pt}{1em}{0.4em}
\titlespacing*{\subsubsection}{0pt}{0.8em}{0.3em}

\usepackage[titles]{tocloft}
\renewcommand{\cftsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftsecaftersnum}{.}
\renewcommand{\cftsubsecaftersnum}{.}

% Captions: tables above, figures below.
\usepackage{caption}
\DeclareCaptionLabelFormat{tablelabel}{Таблица #2}
\captionsetup[table]{labelformat=tablelabel, labelsep=period, position=above, font={small}, labelfont={bf}}
\DeclareCaptionLabelFormat{figurelabel}{Фигура #2}
\captionsetup[figure]{labelformat=figurelabel, labelsep=period, position=below, font={small}, labelfont={bf}}
\newcommand{\tabref}[1]{(табл.~\ref{#1})}
\newcommand{\figref}[1]{(фиг.~\ref{#1})}

\usepackage{amsmath}
\usepackage{newtxmath}  % Times-matched figures and symbols in math mode
\usepackage{graphicx}

% Inline code and identifiers.
% An underscore inside an identifier is a legal break point, so long OpenSSL names
% such as SSL_TLSEXT_ERR_ALERT_FATAL do not run into the margin.
\newcommand{\codeunderscore}{\textunderscore\allowbreak}
\DeclareRobustCommand{\code}[1]{{\ttfamily\let\_\codeunderscore #1}}
% Justified text with long Latin identifiers needs room to reflow.
\setlength{\emergencystretch}{3em}

% Tables, listings and figures at single spacing.
\usepackage{etoolbox}
\AtBeginEnvironment{tabular}{\setstretch{1.1}}
\AtBeginEnvironment{tabularx}{\setstretch{1.1}}
\AtBeginEnvironment{longtable}{\setstretch{1.1}}
\AtBeginEnvironment{tikzpicture}{\setstretch{1.0}}

% C++ listings, read straight from the repository so the text cannot drift from the code.
\usepackage{listings}
\renewcommand{\lstlistingname}{Листинг}
\definecolor{kw}{RGB}{0,0,150}
\definecolor{cm}{RGB}{0,110,0}
\definecolor{st}{RGB}{160,0,0}
\definecolor{ln}{RGB}{120,120,120}
\lstset{
  language=C++, basicstyle=\footnotesize\ttfamily\setstretch{1.0}, columns=fullflexible, keepspaces=true,
  keywordstyle=\color{kw}\bfseries, commentstyle=\color{cm}, stringstyle=\color{st},
  numbers=left, numberstyle=\tiny\color{ln}, numbersep=6pt, xleftmargin=2em,
  frame=single, rulecolor=\color{ln}, framerule=0.4pt, breaklines=true,
  postbreak=\mbox{\textcolor{ln}{$\hookrightarrow$}\space}, tabsize=4, showstringspaces=false,
  captionpos=b, aboveskip=0.8em, belowskip=0.6em, inputencoding=utf8,
  morekeywords={constexpr, noexcept, nullptr, override, final, span, string_view, expected, unexpected},
}
